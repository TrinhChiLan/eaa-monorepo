import aiosqlite
import json

class DatabaseRepository:
    def __init__(self, db_path: str = "edge_system.db"):
        self.db_path = db_path
        self.db_connection = None

    async def initialize(self):
        """Initializes the SQLite database connection and sets WAL mode for performance."""
        self.db_connection = await aiosqlite.connect(self.db_path)
        
        # Enable Write-Ahead Logging (WAL) mode for non-blocking concurrent read/write
        await self.db_connection.execute("PRAGMA journal_mode=WAL;")
        await self.db_connection.commit()
        
        # Create tables if they don't exist
        await self.create_tables()

    async def create_tables(self):
        """Creates required tables for telemetry logs and events."""
        async with self.db_connection.cursor() as cursor:
            await cursor.execute("""
                CREATE TABLE IF NOT EXISTS telemetry_logs (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    timestamp REAL,
                    payload TEXT
                )
            """)
            await self.db_connection.commit()

    async def bulk_insert_telemetry(self, items: list):
        """Efficiently writes a batch of telemetry items to the database."""
        if not items:
            return

        # Prepare records for insertion (converting Pydantic models/dicts to JSON strings)
        records = [
            (getattr(item, 'timestamp', 0.0), item.model_dump_json() if hasattr(item, 'model_dump_json') else json.dumps(item))
            for item in items
        ]

        async with self.db_connection.cursor() as cursor:
            await cursor.executemany(
                "INSERT INTO telemetry_logs (timestamp, payload) VALUES (?, ?)",
                records
            )
            await self.db_connection.commit()

    async def close(self):
        """Safely closes the database connection."""
        if self.db_connection:
            await self.db_connection.close()
